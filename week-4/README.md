## Задача 1

Какво се случва на всеки ред?

```
#include <iostream>

int main() {
    int a = 10, b = 20;
    int& ref = a;
    int* ptr = &b;

    ref = 15;
    *ptr = ref + 5;

    ptr = &a;
    *ptr += 10;

    ref = b;
    ref += 5;


    return 0;
}
```

Какво ще се изведе на екрана?

```
#include <iostream>

void mystery(int a, int& b, int* c, int* d) {
    a += 10;
    b += 10;
    *c += 10;
    d = &a;
    *d += 100;
}

int main() {
    int x = 1, y = 2, z = 3, w = 4;
    mystery(x, y, &z, &w);
    std::cout << x << " " << y << " " << z << " " << w << "\n";
    return 0;
}
```
